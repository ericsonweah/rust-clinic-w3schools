# 🛠 Step 2 — Parse headers & body in `handle_connection`

Update the request parsing logic:

```rust
fn parse_request(buffer: &[u8]) -> HttpRequest {
    let req_str = String::from_utf8_lossy(buffer).to_string();
    let mut lines = req_str.split("\r\n");

    // Request line
    let request_line = lines.next().unwrap_or("");
    let parts: Vec<&str> = request_line.split_whitespace().collect();
    let method = parts.get(0).unwrap_or(&"GET").to_string();
    let path = parts.get(1).unwrap_or(&"/").to_string();

    // Headers
    let mut headers = HashMap::new();
    for line in &mut lines {
        if line.is_empty() {
            break; // stop at blank line
        }
        if let Some((k, v)) = line.split_once(":") {
            headers.insert(k.trim().to_string(), v.trim().to_string());
        }
    }

    // Body
    let body_raw: String = lines.collect::<Vec<&str>>().join("\n");
    let mut body = HashMap::new();

    if let Some(ct) = headers.get("Content-Type") {
        if ct.contains("application/x-www-form-urlencoded") {
            for pair in body_raw.split('&') {
                if let Some((k, v)) = pair.split_once('=') {
                    body.insert(k.to_string(), v.to_string());
                }
            }
        } else if ct.contains("application/json") {
            // ✅ simplest version: naive JSON parsing
            let trimmed = body_raw.trim().trim_matches(|c| c == '{' || c == '}');
            for pair in trimmed.split(',') {
                if let Some((k, v)) = pair.split_once(':') {
                    body.insert(
                        k.trim().trim_matches('"').to_string(),
                        v.trim().trim_matches('"').to_string(),
                    );
                }
            }
        }
    }

    HttpRequest {
        method,
        path,
        params: HashMap::new(),
        headers,
        body_raw,
        body,
    }
}
```

# ✅ Example Routes

### 1. Raw body

```rust
app.post("/echo", |req| {
    res::ok(&format!("Raw body: {}", req.body_raw))
});
```

```bash
curl -X POST http://127.0.0.1:3000/echo -d "Hello World"
```

➡️ `Raw body: Hello World`

---

### 2. Form data

```rust
app.post("/login", |req| {
    let username = req.body.get("username").unwrap_or(&"".into());
    let password = req.body.get("password").unwrap_or(&"".into());
    res::ok(&format!("Login -> user: {}, pass: {}", username, password))
});
```

```bash
curl -X POST http://127.0.0.1:3000/login \
  -H "Content-Type: application/x-www-form-urlencoded" \
  -d "username=ericson&password=1234"
```

➡️ `Login -> user: ericson, pass: 1234`

---

### 3. JSON

```rust
app.post("/api", |req| {
    let msg = req.body.get("msg").unwrap_or(&"".into());
    res::json(&format!(r#"{{"echo":"{}"}}"#, msg))
});
```

```bash
curl -X POST http://127.0.0.1:3000/api \
  -H "Content-Type: application/json" \
  -d '{"msg":"Hello JSON"}'
```

➡️ `{"echo":"Hello JSON"}`
